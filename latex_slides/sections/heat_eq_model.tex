\begin{frame}
    \frametitle{Heat equation model with nonlinear heat-diffusion coefficient}
    We divide a 1D rod with length $L$ into $N_x$ FV cells of length $\Delta x = \frac{L}{N_x}$.\\
    System equations:
    \begin{align}
        d T_i &= \frac{K_i}{\Delta x^2} (T_{i+1} - 2 T_i + T_{i-1}) dt + g_i dw_i, \quad i = 1, \ldots, N_x \\
        y_k &= C T(t_k) + v_k, \quad v_k \sim \mathcal{N}(0, \sigma_y^2 I)
    \end{align}
    where 
    \begin{align}
        K_i &= K_0 \exp\left( -\beta T_i \right), \\ g_i &= \sigma_T.
    \end{align}
    We choose sensor positions at: 0.1$L$, 0.3$L$, 0.7$L$, and 0.9$L$.\\
    BCs: $T_0 = 0$, $T_{N_x+1} = 10$.\\
    Parameters: $K_0 = 1.0$, $\beta = 0.6$, $\sigma_T = 0.2$, $\sigma_y = 0.05$.

\end{frame}
